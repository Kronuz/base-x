# base_x.hh's leading_zeros flag: the count of zero bytes on encode and of zero digits on decode,
# where the zero digits go, what the checksum and padding cover, and which presets carry the flag.

# --- count on encode

repl: \t\t\twhile (false && zeros < decoded_size && !decoded[zeros]) {\n\t\t\t\t++zeros;\n\t\t\t}

what: encode: the count stops one byte early
find: \t\t\twhile (zeros < decoded_size && !decoded[zeros]) {
repl: \t\t\twhile (zeros + 1 < decoded_size && !decoded[zeros]) {

mutant: enc-count-one-more
what: encode: the count is one too many
find: \t\t\tencode_value(result, rest ? uinteger_t(decoded + zeros, rest, 256) : uinteger_t(), 0, zeros);
repl: \t\t\tencode_value(result, rest ? uinteger_t(decoded + zeros, rest, 256) : uinteger_t(), 0, zeros + 1);

mutant: enc-count-stops-at-one

repl: \t\t\tencode_value(result, rest ? uinteger_t(decoded + zeros, rest, 256) : uinteger_t(), 0, zeros ? 1 : 0);

# --- count on decode

mutant: dec-number-after
